   270→// Check if streak should be reset (more than 1 day since last lesson)
   271→function computeStreakOnLoad(state: GamificationState): GamificationState {
   272→  if (!state.lastLessonAt) return state;
   273→  const now = Date.now();
   274→  const dayMs = 24 * 60 * 60 * 1000;
   275→  const daysSince = Math.floor((now - state.lastLessonAt) / dayMs);
   276→  if (daysSince <= 1) return state; // still active
   277→  if (state.streakFreezes > 0 && daysSince === 2) {
   278→    return { ...state, streakFreezes: state.streakFreezes - 1 };
   279→  }
   280→  return { ...state, currentStreak: 0, dailyXP: 0 };
   281→}

   308→
   309→      setUser: (user) => set({ user }),
   310→
   311→      loginDemo: () =>
   312→        set({
   313→          user: {
   314→            id: "demo-user",
   315→            email: "demo@lingoland.app",
   316→            name: "Lumo Learner",
   317→            avatar: "🦊",
   318→            country: "🌍",
   319→            role: "learner",
   320→            isAuthenticated: true,
   321→            createdAt: Date.now(),
   322→          },
   323→          view: "home",
   324→        }),
   325→
   326→      logout: () => set({ user: null, view: "landing" }),
   327→
   328→      setActiveCourse: (courseId) =>
   329→        set((s) => {
   330→          if (!s.coursesProgress[courseId]) {
   331→            const course = COURSES.find((c) => c.id === courseId);
   332→            s.coursesProgress[courseId] = {
   333→              courseId,
   334→              completedLessons: [],
   335→              lessonStars: {},
   336→              currentUnitOrder: 1,
   337→              currentLessonOrder: 1,
   338→              lastVisitedAt: null,
   339→            };
   340→            void course;
   341→          }
   342→          return { activeCourseId: courseId };
   343→        }),
   344→
   345→      startLesson: (lesson) =>
   346→        set({
   347→          view: "lesson",
   348→          activeLesson: {
   349→            lessonId: lesson.id,
   350→            exercises: lesson.exercises,
   351→            currentIndex: 0,
   352→            correctCount: 0,
   353→            wrongCount: 0,
   354→            earnedXP: 0,
   355→            answers: [],
   356→            startedAt: Date.now(),
   357→          },
   358→        }),
   359→
   360→      recordAnswer: (exerciseId, correct, userAnswer, xpEarned) =>
   361→        set((s) => {
   362→          if (!s.activeLesson) return {};
   363→          const al = { ...s.activeLesson };
   364→          al.answers = [...al.answers, { exerciseId, correct, userAnswer }];
   365→          if (correct) {
   366→            al.correctCount += 1;
   367→            al.earnedXP += xpEarned;
   368→          } else {
   369→            al.wrongCount += 1;
   370→          }
   371→          return { activeLesson: al };
   372→        }),
   373→
   374→      advanceLesson: () =>
   375→        set((s) => {
   376→          if (!s.activeLesson) return {};
   377→          return { activeLesson: { ...s.activeLesson, currentIndex: s.activeLesson.currentIndex + 1 } };
   378→        }),
   379→
   380→      completeLesson: () => {
   381→        const s = get();
   382→        if (!s.activeLesson) return { xpEarned: 0, perfect: false, stars: 0 };
   383→        const al = s.activeLesson;
   384→        const total = al.exercises.length;
   385→        const correct = al.correctCount;
   386→        const perfect = al.wrongCount === 0;
   387→        const stars = perfect ? 3 : al.wrongCount <= 1 ? 2 : 1;
   388→        const bonus = perfect ? Math.floor(al.earnedXP * 0.5) : 0;
   389→        const xpEarned = al.earnedXP + bonus;
   390→        // Update gamification
   391→        const now = Date.now();
   392→        const dayMs = 24 * 60 * 60 * 1000;
   393→        const lastTs = s.gamification.lastLessonAt;
   394→        const newStreak = lastTs && now - lastTs < dayMs * 2 ? s.gamification.currentStreak + 1 : 1;
   395→        const courseProgress = s.activeCourseId ? s.coursesProgress[s.activeCourseId] : null;
   396→        const newCoursesProgress = { ...s.coursesProgress };
   397→        if (s.activeCourseId && courseProgress) {
   398→          const cp = { ...courseProgress };
   399→          if (!cp.completedLessons.includes(al.lessonId)) {
   400→            cp.completedLessons = [...cp.completedLessons, al.lessonId];
   401→          }
   402→          cp.lessonStars = { ...cp.lessonStars, [al.lessonId]: Math.max(cp.lessonStars[al.lessonId] ?? 0, stars) };
   403→          // Advance pointer
   404→          const course = COURSES.find((c) => c.id === s.activeCourseId);
   405→          if (course) {
   406→            for (const unit of course.units) {
   407→              const idx = unit.lessons.findIndex((l) => l.id === al.lessonId);
   408→              if (idx >= 0) {
   409→                if (idx + 1 < unit.lessons.length) {
   410→                  cp.currentLessonOrder = idx + 2; // 1-indexed
   411→                  cp.currentUnitOrder = unit.order || 1;
   412→                } else {
   413→                  // Move to next unit
   414→                  const uIdx = course.units.findIndex((u) => u.id === unit.id);
   415→                  if (uIdx + 1 < course.units.length) {
   416→                    cp.currentUnitOrder = (course.units[uIdx + 1].order) || (uIdx + 2);
   417→                    cp.currentLessonOrder = 1;
   418→                  }
   419→                }
   420→                break;
   421→              }
   422→            }
   423→          }
   424→          cp.lastVisitedAt = now;
   425→          newCoursesProgress[s.activeCourseId] = cp;
   426→        }
   427→        const newGamification: GamificationState = {
   428→          ...s.gamification,
   429→          totalXP: s.gamification.totalXP + xpEarned,
   430→          weeklyXP: s.gamification.weeklyXP + xpEarned,
   431→          dailyXP: s.gamification.dailyXP + xpEarned,
   432→          lastLessonAt: now,
   433→          currentStreak: newStreak,
   434→          longestStreak: Math.max(s.gamification.longestStreak, newStreak),
   435→          hearts: perfect ? s.gamification.hearts : Math.min(s.gamification.maxHearts, s.gamification.hearts),
   436→        };
   437→        // Auto-unlock achievements
   438→        const newAchievements = { ...s.achievementsProgress };
   439→        const newNotifs: NotificationItem[] = [];
   440→        for (const ach of ACHIEVEMENTS) {
   441→          const already = newAchievements[ach.id]?.unlockedAt;
   442→          if (already) continue;
   443→          let value = 0;
   444→          if (ach.category === "streak") value = newGamification.currentStreak;
   445→          else if (ach.category === "xp") value = newGamification.totalXP;
   446→          else if (ach.category === "lesson") {
   447→            // Count completed lessons (with id prefixes)
   448→            const totalCompleted = Object.values(newCoursesProgress).reduce((acc, p) => acc + p.completedLessons.length, 0);
   449→            value = totalCompleted;
   450→            if (ach.id === "perfect-1") value = perfect ? 1 : (s.achievementsProgress["perfect-1"]?.unlockedAt ? 1 : 0);
   451→            if (ach.id === "perfect-10") value = 0; // we don't track historical; skip
   452→          } else if (ach.category === "skill") {
   453→            if (ach.id === "vocab-100" || ach.id === "vocab-500") value = s.reviewItems.length;
   454→            if (ach.id === "review-50") value = s.reviewItems.reduce((acc, r) => acc + r.correctCount, 0);
   455→          } else if (ach.category === "social") {
   456→            if (ach.id.startsWith("friend")) value = 0;
   457→            if (ach.id === "leaderboard-1") value = 0;
   458→          } else if (ach.category === "special") {
   459→            if (ach.id.startsWith("ai")) value = s.aiConversations.length;
   460→            if (ach.id.startsWith("quest")) {
   461→              value = Object.values(s.questsProgress).filter((q) => q.completed).length;
   462→            }
   463→          }
   464→          if (value >= ach.requirement) {
   465→            newAchievements[ach.id] = { unlockedAt: now };
   466→            newNotifs.push({
   467→              id: genId("ntf"),
   468→              type: "achievement",
   469→              title: `Achievement unlocked: ${ach.title}!`,
   470→              body: ach.description,
   471→              icon: ach.icon,
   472→              link: "achievements",
   473→              read: false,
   474→              createdAt: now,
   475→            });
   476→          }
   477→        }
   478→        // Progress quests
   479→        const newQuests = { ...s.questsProgress };
   480→        for (const q of QUESTS) {
   481→          const qp = newQuests[q.id] ?? { id: q.id, progress: 0, completed: false, claimed: false };
   482→          if (qp.completed) continue;
   483→          if (q.goalType === "xp" || q.goalType === "perfect") qp.progress += xpEarned;
   484→          if (q.goalType === "lessons") qp.progress += 1;
   485→          if (q.goalType === "perfect" && perfect) qp.progress += 1;
   486→          if (qp.progress >= q.goalAmount) {
   487→            qp.progress = q.goalAmount;
   488→            qp.completed = true;
   489→            newNotifs.push({
   490→              id: genId("ntf"),
   491→              type: "quest",
   492→              title: `Quest complete: ${q.title}`,
   493→              body: `Tap to claim ${q.xp} XP and ${q.gems} gems.`,
   494→              icon: q.icon,
   495→              link: "quests",
   496→              read: false,
   497→              createdAt: now,
   498→            });
   499→          }
   500→          newQuests[q.id] = { ...qp };
   501→        }
   502→        set({
   503→          gamification: newGamification,
   504→          coursesProgress: newCoursesProgress,
   505→          achievementsProgress: newAchievements,
   506→          questsProgress: newQuests,
   507→          notifications: [...newNotifs, ...s.notifications],
   508→          view: "lesson-complete",
   509→          activeLesson: { ...al, earnedXP: xpEarned },
   510→        });
   511→        return { xpEarned, perfect, stars };
   512→      },
   513→
   514→      quitLesson: () => set({ activeLesson: null, view: "learn" }),
   515→
   516→      addXP: (amount, reason) =>

   519→            ...s.gamification,
   520→            totalXP: s.gamification.totalXP + amount,

   522→            dailyXP: s.gamification.dailyXP + amount,

   540→      },
   541→
   542→      buyShopItem: (itemId, priceGems, name) => {
   543→        const s = get();
   544→        if (s.gamification.gems < priceGems) return false;
   545→        const newNotif: NotificationItem = {

   547→          type: "system",
   548→          title: `Purchased: ${name}`,
   549→          body: "Your item is ready in your profile.",
   550→          icon: "🛍️",
   551→          link: "profile",
   552→          read: false,
   553→          createdAt: Date.now(),
   554→        };
   555→        // Apply item effects
   556→        let g = { ...s.gamification, gems: s.gamification.gems - priceGems };
   557→        if (itemId === "heart-refill") g.hearts = g.maxHearts;
   558→        if (itemId === "heart-1") g.hearts = Math.min(g.maxHearts, g.hearts + 1);
   559→        if (itemId === "streak-freeze") g.streakFreezes += 1;
   560→        if (itemId === "boost-2x-15") g.boosts = [...g.boosts, { type: "2x_xp", expiresAt: Date.now() + 15 * 60 * 1000 }];
   561→        if (itemId === "boost-perfect-1") g.boosts = [...g.boosts, { type: "perfect_double", expiresAt: Date.now() + 60 * 60 * 1000 }];

   597→            const interval = correct ? Math.max(1, Math.round(r.intervalDays * ease)) : 1;
   598→            return {
   599→              ...r,
   600→              easeFactor: ease,
   601→              repetitions: reps,
   602→              intervalDays: interval,
   603→              correctCount: r.correctCount + (correct ? 1 : 0),
   604→              wrongCount: r.wrongCount + (correct ? 0 : 1),
   605→              strength: Math.max(0, Math.min(1, r.strength + (correct ? 0.15 : -0.3))),
   606→              lastReviewAt: Date.now(),
   607→              nextReviewAt: Date.now() + interval * 24 * 60 * 60 * 1000,
   608→            };
   609→          });

   611→        }),

   649→              title: "Achievement unlocked!",

   652→              link: "achievements",
   653→              read: false,
   654→              createdAt: now,

   663→      progressQuest: (goalType, amount) =>

   670→            const qp = newQuests[q.id] ?? { id: q.id, progress: 0, completed: false, claimed: false };
   671→            if (qp.completed) continue;
   672→            qp.progress = Math.min(q.goalAmount, qp.progress + amount);
   673→            if (qp.progress >= q.goalAmount && !qp.completed) {

   676→                id: genId("ntf"),
   677→                type: "quest",
   678→                title: `Quest complete: ${q.title}`,
   679→                body: `Tap to claim ${q.xp} XP and ${q.gems} gems.`,
   680→                icon: q.icon,
   681→                link: "quests",

   683→                createdAt: now,
   684→              });
   685→            }
   686→            newQuests[q.id] = { ...qp };

   695→          const q = QUESTS.find((qq) => qq.id === id);

   701→              totalXP: s.gamification.totalXP + q.xp,
   702→              weeklyXP: s.gamification.weeklyXP + q.xp,
   703→              dailyXP: s.gamification.dailyXP + q.xp,

   732→

   741→        }),
   742→
   743→      dismissToast: (id) => set((s) => ({ toasts: s.toasts.filter((t) => t.id !== id) })),
   744→
   745→      addNotification: (n) =>
   746→        set((s) => ({
   747→          notifications: [{ ...n, id: genId("ntf"), read: false, createdAt: Date.now() }, ...s.notifications],
   748→        })),
   749→
   750→      markNotificationRead: (id) =>
   751→        set((s) => ({
   752→          notifications: s.notifications.map((n) => (n.id === id ? { ...n, read: true } : n)),
   753→        })),
   754→
   755→      setFocusMode: (on) => set({ focusMode: on }),

   757→      setTheme: (theme) => set({ theme }),
   758→
   759→      resetProgress: () =>

   792→/** Helper: leaderboard data with current user mixed in */
   793→export function useLeaderboard() {
   794→  const me = useApp((s) => s.gamification.weeklyXP);
   795→  const myName = useApp((s) => s.user?.name ?? "You");
   796→  const myAvatar = useApp((s) => s.user?.avatar ?? "🦊");
   797→  const myLeague = useApp((s) => s.gamification.league);
   798→  // Generate consistent weekly XP for bots based on time of week
   799→  const now = new Date();
   800→  const weekStart = new Date(now);
   801→  weekStart.setHours(0, 0, 0, 0);
   802→  weekStart.setDate(now.getDate() - now.getDay());
   803→  const dayOfWeek = (now.getTime() - weekStart.getTime()) / (24 * 60 * 60 * 1000);
   804→  const roster = BOT_LEARNERS.map((b, i) => ({
   805→    name: b.name,
   806→    avatar: b.avatar,
   807→    country: b.country,
   808→    xp: Math.max(20, Math.round(b.baseXP * (0.4 + dayOfWeek / 7 * 0.6) + Math.sin(i * 1.3) * 30)),
   809→    isMe: false,
   810→  }));
   811→  roster.push({ name: myName, avatar: myAvatar, country: "🌍", xp: me, isMe: true });
   812→  roster.sort((a, b) => b.xp - a.xp);
   813→  return { roster, league: myLeague };
   814→}

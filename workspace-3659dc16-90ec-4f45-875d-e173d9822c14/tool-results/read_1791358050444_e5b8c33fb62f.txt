     1→/**
     2→ * Lingoland content catalog.
     3→ * 3 languages (Spanish, Japanese, French), each with multiple units and lessons.
     4→ * All content is original — written specifically for Lingoland.
     5→ *
     6→ * The exercise engine consumes ExerciseSpec objects; lessons are lists of specs.
     7→ */
     8→
     9→export type ExerciseType =
    10→  | "multiple_choice"
    11→  | "translation"
    12→  | "word_order"
    13→  | "fill_blank"
    14→  | "matching"
    15→  | "listening"
    16→  | "speaking"
    17→  | "image_select"
    18→  | "dialogue"
    19→  | "spell"
    20→  | "true_false";
    21→
    22→export interface ExerciseSpec {
    23→  id: string;
    24→  type: ExerciseType;
    25→  /** Translation direction: "to_target" (translate INTO target language) | "to_native" | "listen" | "see" */
    26→  prompt: string;
    27→  promptTranslation?: string;
    28→  instruction?: string;
    29→  /** For multiple_choice, image_select, true_false */
    30→  options?: string[];
    31→  optionTranslations?: string[];
    32→  /** Index(es) of correct option(s) for MC/image/true_false */
    33→  correctIndex?: number | number[];
    34→  /** For word_order — tokens to arrange */
    35→  tokens?: string[];
    36→  tokensTranslation?: string;
    37→  /** Correct token order (as joined string for comparison) */
    38→  correctOrder?: string;
    39→  /** For fill_blank — sentence with __BLANK__ */
    40→  sentence?: string;
    41→  blankAnswer?: string;
    42→  /** For matching — pairs */
    43→  pairs?: { left: string; right: string }[];
    44→  /** For listening */
    45→  audioText?: string;
    46→  /** For speaking */
    47→  targetPhrase?: string;
    48→  /** Hint shown after wrong answer */
    49→  hint?: string;
    50→  /** Explanation shown after answering */
    51→  explanation?: string;
    52→  /** Image url for image_select (we use emoji proxy for offline) */
    53→  images?: string[]; // emoji
    54→  correctImageIndex?: number;
    55→  /** Reward */
    56→  xp?: number;
    57→}

    59→export interface VocabItem {
    60→  word: string;
    61→  translation: string;
    62→  pronunciation?: string;
    63→  partOfSpeech?: string;
    64→  example?: string;
    65→  exampleTranslation?: string;
    66→  difficulty?: number;
    67→  emoji?: string;
    68→}

    70→export interface LessonSpec {
    71→  id: string;
    72→  title: string;
    73→  description: string;
    74→  type: "lesson" | "story" | "practice" | "challenge" | "review" | "speaking";
    75→  xp: number;
    76→  vocab?: VocabItem[];
    77→  exercises: ExerciseSpec[];
    78→  isBonus?: boolean;
    79→}

  1541→export function findCourse(courseId: string): CourseSpec | undefined {
  1542→  return COURSES.find((c) => c.id === courseId);
  1543→}
  1544→
  1545→export function findLesson(lessonId: string): { lesson: LessonSpec; course: CourseSpec; unit: UnitSpec } | undefined {
  1546→  for (const course of COURSES) {
  1547→    for (const unit of course.units) {
  1548→      const lesson = unit.lessons.find((l) => l.id === lessonId);
  1549→      if (lesson) return { lesson, course, unit };
  1550→    }
  1551→  }
  1552→  return undefined;
  1553→}

  1555→/** Flatten vocab across a unit for SRS / review */

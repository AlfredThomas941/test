   270→   270→// Check if streak should be reset (more than 1 day since last lesson)
   271→   271→function computeStreakOnLoad(state: GamificationState): GamificationState {
   272→   272→  if (!state.lastLessonAt) return state;
   273→   273→  const now = Date.now();
   274→   274→  const dayMs = 24 * 60 * 60 * 1000;
   275→   275→  const daysSince = Math.floor((now - state.lastLessonAt) / dayMs);
   276→   276→  if (daysSince <= 1) return state; // still active
   277→   277→  if (state.streakFreezes > 0 && daysSince === 2) {
   278→   278→    return { ...state, streakFreezes: state.streakFreezes - 1 };
   279→   279→  }
   280→   280→  return { ...state, currentStreak: 0, dailyXP: 0 };
   281→   281→}

   363→   363→          const al = { ...s.activeLesson };

   375→   375→        set((s) => {

   380→   380→      completeLesson: () => {
   381→   381→        const s = get();
   382→   382→        if (!s.activeLesson) return { xpEarned: 0, perfect: false, stars: 0 };
   383→   383→        const al = s.activeLesson;
   384→   384→        const total = al.exercises.length;
   385→   385→        const correct = al.correctCount;
   386→   386→        const perfect = al.wrongCount === 0;
   387→   387→        const stars = perfect ? 3 : al.wrongCount <= 1 ? 2 : 1;
   388→   388→        const bonus = perfect ? Math.floor(al.earnedXP * 0.5) : 0;
   389→   389→        const xpEarned = al.earnedXP + bonus;
   390→   390→        // Update gamification
   391→   391→        const now = Date.now();
   392→   392→        const dayMs = 24 * 60 * 60 * 1000;
   393→   393→        const lastTs = s.gamification.lastLessonAt;
   394→   394→        const newStreak = lastTs && now - lastTs < dayMs * 2 ? s.gamification.currentStreak + 1 : 1;
   395→   395→        const courseProgress = s.activeCourseId ? s.coursesProgress[s.activeCourseId] : null;
   396→   396→        const newCoursesProgress = { ...s.coursesProgress };
   397→   397→        if (s.activeCourseId && courseProgress) {
   398→   398→          const cp = { ...courseProgress };
   399→   399→          if (!cp.completedLessons.includes(al.lessonId)) {
   400→   400→            cp.completedLessons = [...cp.completedLessons, al.lessonId];
   401→   401→          }
   402→   402→          cp.lessonStars = { ...cp.lessonStars, [al.lessonId]: Math.max(cp.lessonStars[al.lessonId] ?? 0, stars) };
   403→   403→          // Advance pointer
   404→   404→          const course = COURSES.find((c) => c.id === s.activeCourseId);
   405→   405→          if (course) {
   406→   406→            for (const unit of course.units) {
   407→   407→              const idx = unit.lessons.findIndex((l) => l.id === al.lessonId);
   408→   408→              if (idx >= 0) {
   409→   409→                if (idx + 1 < unit.lessons.length) {
   410→   410→                  cp.currentLessonOrder = idx + 2; // 1-indexed
   411→   411→                  cp.currentUnitOrder = unit.order || 1;
   412→   412→                } else {
   413→   413→                  // Move to next unit
   414→   414→                  const uIdx = course.units.findIndex((u) => u.id === unit.id);
   415→   415→                  if (uIdx + 1 < course.units.length) {
   416→   416→                    cp.currentUnitOrder = (course.units[uIdx + 1].order) || (uIdx + 2);
   417→   417→                    cp.currentLessonOrder = 1;
   418→   418→                  }
   419→   419→                }
   420→   420→                break;
   421→   421→              }
   422→   422→            }
   423→   423→          }
   424→   424→          cp.lastVisitedAt = now;
   425→   425→          newCoursesProgress[s.activeCourseId] = cp;
   426→   426→        }
   427→   427→        const newGamification: GamificationState = {
   428→   428→          ...s.gamification,
   429→   429→          totalXP: s.gamification.totalXP + xpEarned,
   430→   430→          weeklyXP: s.gamification.weeklyXP + xpEarned,
   431→   431→          dailyXP: s.gamification.dailyXP + xpEarned,
   432→   432→          lastLessonAt: now,
   433→   433→          currentStreak: newStreak,
   434→   434→          longestStreak: Math.max(s.gamification.longestStreak, newStreak),
   435→   435→          hearts: perfect ? s.gamification.hearts : Math.min(s.gamification.maxHearts, s.gamification.hearts),
   436→   436→        };
   437→   437→        // Auto-unlock achievements
   438→   438→        const newAchievements = { ...s.achievementsProgress };
   439→   439→        const newNotifs: NotificationItem[] = [];
   440→   440→        for (const ach of ACHIEVEMENTS) {
   441→   441→          const already = newAchievements[ach.id]?.unlockedAt;
   442→   442→          if (already) continue;
   443→   443→          let value = 0;
   444→   444→          if (ach.category === "streak") value = newGamification.currentStreak;
   445→   445→          else if (ach.category === "xp") value = newGamification.totalXP;
   446→   446→          else if (ach.category === "lesson") {
   447→   447→            // Count completed lessons (with id prefixes)
   448→   448→            const totalCompleted = Object.values(newCoursesProgress).reduce((acc, p) => acc + p.completedLessons.length, 0);
   449→   449→            value = totalCompleted;
   450→   450→            if (ach.id === "perfect-1") value = perfect ? 1 : (s.achievementsProgress["perfect-1"]?.unlockedAt ? 1 : 0);
   451→   451→            if (ach.id === "perfect-10") value = 0; // we don't track historical; skip
   452→   452→          } else if (ach.category === "skill") {
   453→   453→            if (ach.id === "vocab-100" || ach.id === "vocab-500") value = s.reviewItems.length;
   454→   454→            if (ach.id === "review-50") value = s.reviewItems.reduce((acc, r) => acc + r.correctCount, 0);
   455→   455→          } else if (ach.category === "social") {
   456→   456→            if (ach.id.startsWith("friend")) value = 0;
   457→   457→            if (ach.id === "leaderboard-1") value = 0;
   458→   458→          } else if (ach.category === "special") {
   459→   459→            if (ach.id.startsWith("ai")) value = s.aiConversations.length;
   460→   460→            if (ach.id.startsWith("quest")) {
   461→   461→              value = Object.values(s.questsProgress).filter((q) => q.completed).length;
   462→   462→            }
   463→   463→          }
   464→   464→          if (value >= ach.requirement) {
   465→   465→            newAchievements[ach.id] = { unlockedAt: now };
   466→   466→            newNotifs.push({
   467→   467→              id: genId("ntf"),
   468→   468→              type: "achievement",
   469→   469→              title: `Achievement unlocked: ${ach.title}!`,
   470→   470→              body: ach.description,
   471→   471→              icon: ach.icon,
   472→   472→              link: "achievements",
   473→   473→              read: false,
   474→   474→              createdAt: now,
   475→   475→            });
   476→   476→          }

   792→   792→/** Helper: leaderboard data with current user mixed in */
   793→   793→export function useLeaderboard() {
   794→   794→  const me = useApp((s) => s.gamification.weeklyXP);
   795→   795→  const myName = useApp((s) => s.user?.name ?? "You");
   796→   796→  const myAvatar = useApp((s) => s.user?.avatar ?? "🦊");
   797→   797→  const myLeague = useApp((s) => s.gamification.league);
   798→   798→  // Generate consistent weekly XP for bots based on time of week
   799→   799→  const now = new Date();
   800→   800→  const weekStart = new Date(now);
   801→   801→  weekStart.setHours(0, 0, 0, 0);
   802→   802→  weekStart.setDate(now.getDate() - now.getDay());
   803→   803→  const dayOfWeek = (now.getTime() - weekStart.getTime()) / (24 * 60 * 60 * 1000);
   804→   804→  const roster = BOT_LEARNERS.map((b, i) => ({
   805→   805→    name: b.name,
   806→   806→    avatar: b.avatar,
   807→   807→    country: b.country,
   808→   808→    xp: Math.max(20, Math.round(b.baseXP * (0.4 + dayOfWeek / 7 * 0.6) + Math.sin(i * 1.3) * 30)),
   809→   809→    isMe: false,
   810→   810→  }));
   811→   811→  roster.push({ name: myName, avatar: myAvatar, country: "🌍", xp: me, isMe: true });
   812→   812→  roster.sort((a, b) => b.xp - a.xp);
   813→   813→  return { roster, league: myLeague };
   814→   814→}
